Cloud Auto-Sync

Connect to Your Google Sheet

All new orders, payments, and parties save locally first, then auto-sync to your Google Sheet whenever internet is on.

Connection & Queue Status: Not linked yet
Step 1: Add this Script to your Google Sheet

Open your Google Sheet → Extensions → Apps Script → paste this code → click Deploy → New deployment → Web app (Execute as: Me, Who has access: Anyone).

function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);
  var type = payload.type;
  var d = payload.data;

  if (type === 'order') {
    var sheet = ss.getSheetByName('Orders Register') || ss.insertSheet('Orders Register');
    var row = sheet.getLastRow() + 1;
    sheet.appendRow([
      d.id, d.date, d.seller, d.buyer, d.item, d.qty, d.rate,
      '=F' + row + '*G' + row,
      d.paid || 0,
      d.paid > 0 ? d.date : '',
      '=H' + row + '-I' + row,
      d.paidToSeller || 0,
      '=H' + row + '-L' + row,
      '=IF(K' + row + '<=0,"Paid",IF(I' + row + '>0,"Partial","Pending"))',
      d.remarks || ''
    ]);
  } else if (type === 'party') {
    var idx = ss.getSheetByName('Index') || ss.insertSheet('Index');
    var r = idx.getLastRow() + 1;
    idx.appendRow([d.id, d.name, d.type, d.phone || '', d.city || '']);
  } else if (type === 'payment') {
    var pSheet = ss.getSheetByName('Payments Log') || ss.insertSheet('Payments Log');
    if (pSheet.getLastRow() === 0) {
      pSheet.appendRow(['Payment ID', 'Date', 'Party', 'Direction', 'Amount', 'Mode', 'Linked Order', 'Reference']);
    }
    pSheet.appendRow([d.id, d.date, d.party, d.direction, d.amount, d.mode, d.linkedOrder || '', d.ref || '']);
  }

  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' })).setMimeType(ContentService.MimeType.JSON);
}
