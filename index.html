function doPost(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var payload = JSON.parse(e.postData.contents);
  var type = payload.type;
  var d = payload.data;

  if (type === 'order') {
    var sheet = ss.getSheetByName('Orders Register') || ss.insertSheet('Orders Register');
    var row = sheet.getLastRow() + 1;
    sheet.appendRow([
      d.id, d.date, d.seller, d.buyer, d.item, d.qty, d.rate,
      '=F' + row + '*G' + row,
      d.paid || 0,
      d.paid > 0 ? d.date : '',
      '=H' + row + '-I' + row,
      d.paidToSeller || 0,
      '=H' + row + '-L' + row,
      '=IF(K' + row + '<=0,"Paid",IF(I' + row + '>0,"Partial","Pending"))',
      d.remarks || ''
    ]);
  } else if (type === 'party') {
    var idx = ss.getSheetByName('Index') || ss.insertSheet('Index');
    var r = idx.getLastRow() + 1;
    idx.appendRow([d.id, d.name, d.type, d.phone || '', d.city || '']);
  } else if (type === 'payment') {
    var pSheet = ss.getSheetByName('Payments Log') || ss.insertSheet('Payments Log');
    if (pSheet.getLastRow() === 0) {
      pSheet.appendRow(['Payment ID', 'Date', 'Party', 'Direction', 'Amount', 'Mode', 'Linked Order', 'Reference']);
    }
    pSheet.appendRow([d.id, d.date, d.party, d.direction, d.amount, d.mode, d.linkedOrder || '', d.ref || '']);
  }

  return ContentService.createTextOutput(JSON.stringify({ status: 'ok' })).setMimeType(ContentService.MimeType.JSON);
}

function doGet(e) {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var tz = ss.getSpreadsheetTimeZone();
  function fmtDate(v) {
    if (v instanceof Date) return Utilities.formatDate(v, tz, 'yyyy-MM-dd');
    return v ? String(v) : '';
  }

  var parties = [];
  var idx = ss.getSheetByName('Index');
  if (idx && idx.getLastRow() > 1) {
    var pRows = idx.getRange(2, 1, idx.getLastRow() - 1, 5).getValues();
    for (var i = 0; i < pRows.length; i++) {
      if (pRows[i][1]) {
        parties.push({
          id: String(pRows[i][0] || ('p_' + i)),
          name: String(pRows[i][1]).trim(),
          type: String(pRows[i][2] || 'Retailer').trim(),
          phone: String(pRows[i][3] || ''),
          city: String(pRows[i][4] || '')
        });
      }
    }
  }

  var orders = [];
  var ordSheet = ss.getSheetByName('Orders Register');
  if (ordSheet && ordSheet.getLastRow() > 1) {
    var oRows = ordSheet.getRange(2, 1, ordSheet.getLastRow() - 1, 15).getValues();
    for (var j = 0; j < oRows.length; j++) {
      if (oRows[j][2] || oRows[j][3]) {
        orders.push({
          id: String(oRows[j][0] || ('ORD-' + (1001 + j))),
          date: fmtDate(oRows[j][1]),
          seller: String(oRows[j][2] || '').trim(),
          buyer: String(oRows[j][3] || '').trim(),
          item: String(oRows[j][4] || '').trim(),
          qty: Number(oRows[j][5]) || 0,
          unit: 'Kg',
          rate: Number(oRows[j][6]) || 0,
          amount: Number(oRows[j][7]) || (Number(oRows[j][5]) * Number(oRows[j][6])) || 0,
          paid: Number(oRows[j][8]) || 0,
          paidToSeller: Number(oRows[j][11]) || 0,
          remarks: String(oRows[j][14] || '')
        });
      }
    }
  }

  var payments = [];
  var paySheet = ss.getSheetByName('Payments Log');
  if (paySheet && paySheet.getLastRow() > 1) {
    var payRows = paySheet.getRange(2, 1, paySheet.getLastRow() - 1, 8).getValues();
    for (var k = 0; k < payRows.length; k++) {
      if (payRows[k][2]) {
        payments.push({
          id: String(payRows[k][0] || ('PAY-' + (101 + k))),
          date: fmtDate(payRows[k][1]),
          party: String(payRows[k][2]).trim(),
          direction: String(payRows[k][3] || 'from_buyer'),
          amount: Number(payRows[k][4]) || 0,
          mode: String(payRows[k][5] || 'Cash'),
          linkedOrder: String(payRows[k][6] || 'General Balance'),
          ref: String(payRows[k][7] || '')
        });
      }
    }
  }

  return ContentService.createTextOutput(JSON.stringify({
    status: 'ok',
    parties: parties,
    orders: orders,
    payments: payments
  })).setMimeType(ContentService.MimeType.JSON);
}
